     3→
     4→// ============== Agent Tool Registry ==============
     5→// Each tool has: name, description, parameters (zod), execute function.
     6→// Read tools return data directly. Write tools return a "plan" (proposed mutations)
     7→// that the user must approve before commit.
     8→
     9→import { z } from 'zod'
    10→
    11→export type ToolResult = {
    12→  text?: string
    13→  json?: any
    14→  // For write tools: proposed mutations; if present, the UI shows an approval card.
    15→  plan?: {
    16→    summary: string
    17→    creates: Array<{ table: string; data: any }>
    18→    updates?: Array<{ table: string; id: string; data: any }>
    19→    sideEffects?: string[]
    20→    approvalId?: string
    21→  }
    22→  // For commit step: actually persist
    23→  commit?: () => Promise<any>
    24→}
    25→
    26→export interface AgentTool {
    27→  name: string
    28→  description: string
    29→  domain: string
    30→  isWrite: boolean
    31→  schema: z.ZodObject<any, any>
    32→  execute: (args: any) => Promise<ToolResult>
    33→}
    34→
    35→// helper to safely query masters
    36→async function listAll(model: string, where: any = {}) {
    37→  const m = (db as any)[model]
    38→  if (!m) throw new Error('Unknown model: ' + model)
    39→  // Try orderBy code first, fallback to no orderBy
    40→  try {
    41→    return await m.findMany({ where, take: 100, orderBy: { code: 'asc' } })
    42→  } catch {
    43→    try {
    44→      return await m.findMany({ where, take: 100 })
    45→    } catch {
    46→      return []
    47→    }
    48→  }
    49→}

  1178→function groupBy(arr: any[], path: string) {
  1179→  const out: Record<string, number> = {}
  1180→  for (const item of arr) {
  1181→    const keys = path.split('.')
  1182→    let val: any = item
  1183→    for (const k of keys) val = val?.[k]
  1184→    const key = String(val || 'unknown')
  1185→    out[key] = (out[key] || 0) + (item.qty || 0)
  1186→  }
  1187→  return out
  1188→}
